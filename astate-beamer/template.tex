%% template.tex

\documentclass[aspectratio=169, 14pt]{beamer}

% ---------------------------------------------------------------------------
% Prompt font via fontspec (XeLaTeX/LuaLaTeX required — same engine as icons)
%   If Prompt is not installed, remove the two \set*font lines; fontspec
%   will fall back to the system default sans-serif.
% ---------------------------------------------------------------------------
\usepackage{fontspec}
\IfFontExistsTF{Prompt}{%
  \setmainfont{Prompt}%
  \setsansfont{Prompt}%
}{}

% ---------------------------------------------------------------------------
% Icons  (require XeLaTeX or LuaLaTeX)
%   fontawesome6 — \faPaperPlane, \faGithub, \faLinkedin, \faXTwitter, \faOrcid ...
%   academicons  — loaded as isolated font to avoid encoding conflicts
% ---------------------------------------------------------------------------
\usepackage{fontawesome6}
% NOTE: \usepackage{academicons} is NOT used — it corrupts fontspec encodings
% under XeLaTeX, garbling the main text font.  Instead, load the Academicons
% font in isolation and define only the glyphs we need.
\IfFontExistsTF{Academicons}{%
  \newfontfamily\academiconsFont{Academicons}%
  \newcommand{\aiDataverse}{{\academiconsFont\symbol{"E9F7}}}%
}{% fallback if font is not installed
  \newcommand{\aiDataverse}{\textsc{dv}}%
}
\usepackage{hyperref}

% ---------------------------------------------------------------------------

% ---------------------------------------------------------------------------
